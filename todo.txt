1) Sanitizers (Debug builds)

Stack them by default; flip off in Release.

# CMakeLists.txt (top-level)
option(SIM_ENABLE_ASAN "Enable AddressSanitizer" ON)
option(SIM_ENABLE_UBSAN "Enable UndefinedBehaviorSanitizer" ON)
option(SIM_ENABLE_TSAN "Enable ThreadSanitizer" OFF)

function(sim_add_sanitizers target)
  if (CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(${target} PRIVATE -fno-omit-frame-pointer -g)
    target_link_options(${target} PRIVATE -fno-omit-frame-pointer)
    if (SIM_ENABLE_ASAN)
      target_compile_options(${target} PRIVATE -fsanitize=address -fsanitize-address-use-after-scope)
      target_link_options(${target} PRIVATE -fsanitize=address)
    endif()
    if (SIM_ENABLE_UBSAN)
      target_compile_options(${target} PRIVATE -fsanitize=undefined)
      target_link_options(${target} PRIVATE -fsanitize=undefined)
    endif()
    if (SIM_ENABLE_TSAN)
      target_compile_options(${target} PRIVATE -fsanitize=thread)
      target_link_options(${target} PRIVATE -fsanitize=thread)
    endif()
  endif()
endfunction()

# apply to your binaries/tests
sim_add_sanitizers(simcore_app)
sim_add_sanitizers(simcore_tests)


Usage tips:

Use ASan+UBSan for everyday debug runs.

Switch on TSan when you suspect races (it’s slower, so run focused tests: ctest -R JobQueue).

}

Short answer: your mtest works as-is. None of the new runtime features (budget monitor, phase graph, chunk tuning) require special flags. The only time you’d change the build is when you want SIMD/NUMA/sanitizers/profilers enabled.

If you’d like handy variants, here are drop-in aliases:

# Clean, out-of-source debug build + tests (ASan+UBSan on)
alias mtest='rm -rf build && cmake -B build -S . \
  -DENABLE_TESTS=ON -DCMAKE_BUILD_TYPE=Debug \
  -DSIM_ENABLE_AVX2=ON \
  -DSIM_ENABLE_ASAN=ON -DSIM_ENABLE_UBSAN=ON -DSIM_ENABLE_TSAN=OFF \
  && cmake --build build -j && (cd build && ctest --output-on-failure -j8)'

# NUMA-aware build (requires libnuma dev headers)
alias mtest_numa='rm -rf build && cmake -B build -S . \
  -DENABLE_TESTS=ON -DCMAKE_BUILD_TYPE=Debug \
  -DSIM_ENABLE_AVX2=ON -DSIM_USE_NUMA=ON \
  -DSIM_ENABLE_ASAN=ON -DSIM_ENABLE_UBSAN=ON \
  && cmake --build build -j && (cd build && ctest --output-on-failure -j8)'

# Race-hunting build (ThreadSanitizer). Slower—run focused tests.
alias mtest_tsan='rm -rf build && cmake -B build -S . \
  -DENABLE_TESTS=ON -DCMAKE_BUILD_TYPE=Debug \
  -DSIM_ENABLE_AVX2=OFF \
  -DSIM_ENABLE_ASAN=OFF -DSIM_ENABLE_UBSAN=OFF -DSIM_ENABLE_TSAN=ON \
  && cmake --build build -j && (cd build && ctest --output-on-failure -R JobQueue -j1)'

# Perf build (RelWithDebInfo) for profiling (perf/Tracy), no sanitizers
alias mperf='rm -rf build-prof && cmake -B build-prof -S . \
  -DENABLE_TESTS=ON -DCMAKE_BUILD_TYPE=RelWithDebInfo \
  -DSIM_ENABLE_AVX2=ON -DSIM_ENABLE_ASAN=OFF -DSIM_ENABLE_UBSAN=OFF \
  && cmake --build build-prof -j && (cd build-prof && ctest --output-on-failure)'


Notes:

SIMD: add -DSIM_ENABLE_AVX2=ON when you want AVX2 paths compiled.

NUMA: add -DSIM_USE_NUMA=ON only if you have libnuma installed and plan to use thread pinning/NUMA packing (the on/off usage is still a runtime setting in Settings).

Sanitizers: ASan+UBSan for day-to-day; TSAN only for targeted race hunts (disable AVX2 there to reduce false positives/noise).

Profiler (optional): if you later wire Tracy, flip -DSIM_ENABLE_TRACY=ON in your build alias.

If you’re happy staying in your original in-source style, just tack the desired -D options onto your current alias; otherwise the out-of-source variants above keep things cleaner.
