cmake_minimum_required(VERSION 3.20)
project(benchmark_embedding LANGUAGES C CXX)
if (NOT RTFW_SOURCE_DIR)
    message(FATAL_ERROR "Set RTFW_SOURCE_DIR to the source checkout")
endif()
set(RTFW_BUILD_BENCHMARKS ON CACHE BOOL "" FORCE)
set(RTFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(RTFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(RTFW_BUILD_RUNTIME_DEMO OFF CACHE BOOL "" FORCE)
set(RTFW_BUILD_EXPERIMENTAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${RTFW_SOURCE_DIR}" rtfw-build EXCLUDE_FROM_ALL)
add_executable(benchmark_embedding "${CMAKE_CURRENT_LIST_DIR}/../../package_consumer/benchmark_consumer.cpp")
target_link_libraries(benchmark_embedding PRIVATE rtfw::benchmark)
enable_testing()
add_test(NAME benchmark_add_subdirectory COMMAND benchmark_embedding)
